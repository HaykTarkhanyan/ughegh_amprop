\begin{tikzpicture}[lbl/.style={font=\small, text=plum, align=center},
    head/.style={font=\blackfont\normalsize, text=plum, align=center},

      \draw[plum, line width=0.9pt] (0,0) ellipse (0.45 and 0.6);
      \node[lbl] at (0,-0.95) {կամար};

      \end{scope}
      \draw[plum, line width=0.9pt] (0,0) ellipse (0.45 and 0.6);
